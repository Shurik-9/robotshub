const WAREHOUSE_SECTOR = 'Транспорт и логистика';
const solution = (() => {
  const found = solutionsData.items.find(item => item.id === DEMO_SOLUTION_ID);
  if (!found || !(typeof found.price_rub === 'number' && Number.isFinite(found.price_rub) && found.price_rub > 0)) {
    throw new Error('Для быстрого расчёта нужен товарный и ценовой источник Ronavi H1500.');
  }
  return found;
})();

function calculateWarehouseEstimate(objectParams: Record<string, number | string | boolean>) {
  const inputs = {
    objectParams,
    solution,
    whatIf: { salaryMultiplier: 1, priceMultiplier: 1, volumeMultiplier: 1 },
    assumptionsOverrides: {},
  };
  const simulation = calculateSimulationInputs(inputs);
  const purchase = calculateEconomics(inputs).find(item => item.id === 'purchase') ?? null;
  return { simulation, purchase };
}

function formatMillions(value: number | null | undefined) {
  return typeof value === 'number' && Number.isFinite(value)
    ? `~${(value / 1_000_000).toLocaleString('ru-RU', { maximumFractionDigits: 2 })} млн ₽`
    : '—';
}

function formatRobotCount(count: number) {
  const mod100 = count % 100;
  const mod10 = count % 10;
  const unit = mod100 >= 11 && mod100 <= 14 ? 'роботов'
    : mod10 === 1 ? 'робот'
      : mod10 >= 2 && mod10 <= 4 ? 'робота'
        : 'роботов';
  return `${count.toLocaleString('ru-RU')} ${unit}`;
}

function formatPaybackForPreview(value: number | null | undefined) {
  if (typeof value !== 'number' || !Number.isFinite(value) || value < 0) {
    return '—';
  }
  const rounded = Math.round(value * 10) / 10;
  const mod100 = rounded % 100;
  const mod10 = rounded % 10;
  const unit = rounded % 1 !== 0 ? 'года'
    : mod100 >= 11 && mod100 <= 14 ? 'лет'
      : mod10 === 1 ? 'год'
        : mod10 >= 2 && mod10 <= 4 ? 'года'
          : 'лет';
  return `${rounded.toLocaleString('ru-RU', { maximumFractionDigits: 1 })} ${unit}`;
}

function WarehouseAssumptions({ area }: { area: number }) {
  const params = estimateObjectParams(area);
  const inbound = Number(params.inbound_pallets_per_day);
  const outbound = Number(params.outbound_pallets_per_day);
  const currency = (value: number) => `${Math.round(value).toLocaleString('ru-RU')} ₽`;
  return (
    <details className="start-assumptions-disclosure" data-testid="start-assumptions-preview">
      <summary>Показать числовые допущения модели</summary>
      <ul>
        <li>
          <strong>Паллетный поток:</strong> {inbound.toLocaleString('ru-RU')} на приёмку и {outbound.toLocaleString('ru-RU')} на отгрузку в сутки.
          Модель выводит его из площади ({area.toLocaleString('ru-RU')} ÷ 20, минимум 20), а производительность считает по входящему потоку без сложения двух направлений. Это не замер площадки.
        </li>
        <li>
          <strong>Режим и персонал:</strong> {params.shifts_per_day} смены по {params.shift_duration_h} ч,
          {' '}{params.working_days_per_year} дней работы в году по модели; {params.pickers_count} комплектовщика по {currency(Number(params.picker_salary_gross))} в месяц,

        </li>
        <li>
          <strong>Планировка для модели:</strong> {Number(params.active_area_m2).toLocaleString('ru-RU')} м² активной зоны,
          {' '}{params.floors_count} этаж, {params.rack_rows} рядов, {params.docks_count} дока и {params.charge_stations} зарядные станции.
        </li>
        <li>

        </li>
      </ul>

  );
}

type PendingWarehouse = {
  sectorId: string;
  area: number;
  objectParams: Record<string, number | string | boolean>;
};

type StartResult = {
  sectorId: string;
  area: number;
  objectParams: PendingWarehouse['objectParams'] | null;
};

function hasSavedProject(state: ProjectState) {
  return !isEmptyProject(state);
}

function makeWarehouseProject(objectParams: PendingWarehouse['objectParams'], demoProject = false): ProjectState {
  return {
    objectType: 'warehouse',
    sectorId: null,
    objectParams,
    selectedSolutions: [DEMO_SOLUTION_ID],
    activeSolutionId: DEMO_SOLUTION_ID,
    whatIfOverrides: {},
    assumptionsOverrides: {},
    financingScenario: 'purchase',
    simulationKpis: null,
    demoProject,
  };
}

          )}

            <button type="button" onClick={() => { setSectorId(null); setResult(null); }}
              className="start-back-link" data-testid="start-change-industry">
              ← Изменить направление
            </button>
          </p>
        </section>
      )}
      </section>

      {result && (
        <section id="start-result" aria-labelledby="start-result-title">
          <h1 id="start-result-title" className="text-2xl font-bold text-foreground sm:text-3xl">
            В базе {selectedIndustry?.n ?? '—'} записей по отрасли «{result.sectorId}»
          </h1>
          <p className="mt-1 text-sm text-muted-foreground">
            {result.area > 0 ? `Площадь склада ${result.area.toLocaleString('ru-RU')} м²` : 'Подбор по отрасли; это не прогноз окупаемости'}
          </p>

          {result.objectParams && calculate ? (
            <>
              <div className="mt-6 grid grid-cols-2 gap-3 lg:grid-cols-4">
                <div className="rounded-xl border border-[rgba(148,163,184,0.16)] bg-[#101724] p-5" data-testid="start-stat-robots">
                  <p className="text-xs uppercase tracking-wider text-muted-foreground">Расчётный парк</p>
                  <p className="mt-2 text-3xl font-bold text-foreground">~{calculate.simulation.robotCount}</p>
                  <p className="mt-1 text-xs text-muted-foreground">роботов по заданному потоку и ТТХ</p>
                </div>
                <div className="rounded-xl border border-[rgba(148,163,184,0.16)] bg-[#101724] p-5" data-testid="start-stat-capex">
                  <p className="text-xs uppercase tracking-wider text-muted-foreground">Оценка CAPEX</p>
                  <p className="mt-2 text-3xl font-bold text-foreground">
                    {formatMillions(calculate.purchase?.capex)}
                  </p>
                  <p className="mt-1 text-xs text-muted-foreground">по модели экономики, не коммерческое предложение</p>
                </div>
                <div className="rounded-xl border border-[rgba(148,163,184,0.16)] bg-[#101724] p-5" data-testid="start-stat-payback">
                  <p className="text-xs uppercase tracking-wider text-muted-foreground">Окупаемость</p>
                  <p className="mt-2 text-3xl font-bold text-foreground">
                    {formatPaybackForPreview(calculate.purchase?.paybackYears ?? null)}
                  </p>
                  <p className="mt-1 text-xs text-muted-foreground">предварительно, по допущениям ниже</p>
                </div>
                <div className="rounded-xl border border-[rgba(148,163,184,0.16)] bg-[#101724] p-5" data-testid="start-stat-flow">
                  <p className="text-xs uppercase tracking-wider text-muted-foreground">Заданный поток</p>
                  <p className="mt-2 text-3xl font-bold text-[#22D3EE]">
                    {calculate.simulation.requiredThroughputPerHour > 0
                      ? `${calculate.simulation.requiredThroughputPerHour.toLocaleString('ru-RU', { maximumFractionDigits: 1 })} палл/ч`
                      : '—'}
                  </p>
                  <p className="mt-1 text-xs text-muted-foreground">рассчитан из допущения об объёме паллет</p>
                </div>
              </div>

              <section className="mt-6 rounded-xl border border-border bg-[#101724] p-4" aria-labelledby="start-assumptions-title" data-testid="start-assumptions">
                <h2 id="start-assumptions-title" className="font-semibold text-foreground">Допущения этой предварительной оценки</h2>
                <p className="mt-2 text-sm leading-relaxed text-muted-foreground">
                  Операции приёмки и отгрузки — по {Number(result.objectParams.inbound_pallets_per_day).toLocaleString('ru-RU')} паллет в сутки каждая; поток в показателе паллет/ч берётся по приёмке, направления не складываются. Режим: {String(result.objectParams.shifts_per_day)} смены по {String(result.objectParams.shift_duration_h)} ч и {String(result.objectParams.working_days_per_year)} дней работы в году по модели. Персонал: {String(result.objectParams.pickers_count)} комплектовщика по {Number(result.objectParams.picker_salary_gross).toLocaleString('ru-RU')} ₽/мес.; коэффициент начислений {Number(result.objectParams.payroll_factor).toLocaleString('ru-RU')}. Планировка: {Number(result.objectParams.active_area_m2).toLocaleString('ru-RU')} м² активной зоны, {String(result.objectParams.floors_count)} этаж, {String(result.objectParams.rack_rows)} рядов, {String(result.objectParams.docks_count)} дока и {String(result.objectParams.charge_stations)} зарядные станции. Это допущения модели, а не сведения о вашем объекте.
                </p>
                <p className="mt-1 text-sm text-muted-foreground">
                  Базовая карточка Ronavi H1500 в каталоге: {(solution.price_rub / 1_000_000).toLocaleString('ru-RU', { maximumFractionDigits: 2 })} млн ₽ за единицу. Итоговый CAPEX выше из-за допущений существующей модели; запросите у поставщика подтверждённую смету.
                </p>
              </section>

              {scene && (
                <div className="mt-6">
                  <h2 className="mb-2 text-lg font-semibold text-foreground">Визуальная модель объекта и парка</h2>
                  <p className="mb-3 text-sm text-muted-foreground">Начальная очередь из {SIMULATION_DEMO_INITIAL_TASKS} заданий нужна только для короткой демонстрации движения. После неё в модели остаётся заданный поток {calculate.simulation.tasksPerHour.toLocaleString('ru-RU', { maximumFractionDigits: 1 })} оп/ч; показатели не являются доказанным прогнозом производительности.</p>
                  <SimCanvas
                    scene={scene}
                    options={{
                      tasksPerHour: calculate.simulation.tasksPerHour,
                      laborCostPerTaskRub: calculate.simulation.laborCostPerTaskRub,
                      batteryRuntimeHours: calculate.simulation.batteryRuntimeHours,
                      chargeTimeMinutes: calculate.simulation.chargeTimeMinutes,
                      initialTasks: SIMULATION_DEMO_INITIAL_TASKS,
                    }}
                    fleet={fleet}
                    sceneKind="warehouse"
                    seed={42}
                    title={`Модель склада ${result.area.toLocaleString('ru-RU')} м² · ${calculate.simulation.robotCount} роб.`}
                    className="min-h-[430px]"
                  />
                </div>
              )}

              <div className="mt-6 grid gap-2 sm:grid-cols-2 lg:grid-cols-4">
                <Link href="/report" className="inline-flex h-11 items-center justify-center gap-2 rounded-md bg-primary px-4 text-sm font-semibold text-primary-foreground hover:bg-primary/90" data-testid="start-open-report">
                  <IconSourceDoc className="h-4 w-4" /> Открыть отчёт
                </Link>
                <Link href="/calc" className="inline-flex h-11 items-center justify-center gap-2 rounded-md border border-border px-4 text-sm font-medium text-foreground hover:bg-secondary" data-testid="start-open-calc">
                  <IconSliders className="h-4 w-4" /> Уточнить расчёт
                </Link>
                <Link href="/simulation" className="inline-flex h-11 items-center justify-center gap-2 rounded-md border border-border px-4 text-sm font-medium text-foreground hover:bg-secondary" data-testid="start-open-sim">
                  <IconClockPayback className="h-4 w-4" /> Открыть симуляцию
                </Link>
                <a href={`/lead?solution=${DEMO_SOLUTION_ID}&via=plan`} className="inline-flex h-11 items-center justify-center gap-2 rounded-md border border-border px-4 text-sm font-medium text-foreground hover:bg-secondary" data-testid="start-request-quote">
                  <IconChartRoi className="h-4 w-4" /> Запросить КП
                </a>
              </div>
            </>
          ) : (
            <>
              <p className="mt-4 max-w-2xl text-sm text-muted-foreground">
                {sectorSolutions.length > 0
                  ? `В каталоге есть ${sectorSolutions.length} решения с указанной ценой. Это подбор, не расчёт окупаемости.`
                  : 'В каталоге не нашлось решений с указанной ценой для этого направления. Это не означает, что их нет на рынке.'}
              </p>
              <div className="mt-5 grid gap-3 sm:grid-cols-3">
                {sectorSolutions.map(item => (
                  <article key={item.id} className="rounded-xl border border-[rgba(148,163,184,0.16)] bg-[#101724] p-4" data-testid="start-sector-solution">
                    <p className="text-sm font-semibold leading-snug text-foreground">{item.name}</p>
                    <p className="mt-1 text-xs text-muted-foreground">{item.vendor}</p>
                    <p className="mt-3 text-sm font-medium text-[#22D3EE]">
                      от {(item.price_rub / 1_000_000).toLocaleString('ru-RU', { maximumFractionDigits: 2 })} млн ₽ по каталогу
                    </p>
                    <p className="mt-1 text-xs text-muted-foreground">{item.status_ru}</p>
                  </article>
                ))}
              </div>
              {similarCases.length > 0 && (
                <div className="mt-6">
                  <h2 className="text-sm font-semibold uppercase tracking-wider text-muted-foreground">Кейсы в этой отрасли</h2>
                  <div className="mt-3 grid gap-3 sm:grid-cols-2">
                    {similarCases.map(item => (
                      <Link key={item.id} href={`/cases/${item.id}`} className="rounded-xl border border-[rgba(148,163,184,0.16)] bg-[#101724] p-4 transition-colors hover:border-[#22D3EE]/50" data-testid="start-sector-case">
                        <p className="text-sm font-semibold text-foreground">{item.title}</p>
                        <p className="mt-1 line-clamp-2 text-xs text-muted-foreground">{item.lead}</p>
                      </Link>
                    ))}
                  </div>
                </div>
              )}
              <div className="mt-6 flex flex-wrap gap-2">
                <Link href={`/quick-select?sector=${encodeURIComponent(result.sectorId)}`} className="inline-flex h-11 items-center rounded-md bg-primary px-5 text-sm font-semibold text-primary-foreground hover:bg-primary/90" data-testid="start-open-catalog">
                  Открыть каталог отрасли
                </Link>
                <Button type="button" variant="outline" className="h-11 px-5" onClick={startDemoCalculation} data-testid="start-demo-calc">
                  Посчитать на демо-складе 5 200 м²
                </Button>
              </div>
            </>
          )}

          <p className="mt-8 text-xs leading-relaxed text-muted-foreground">
            Это предварительная оценка на основе явно перечисленных допущений и каталожных цен. Она не учитывает обследование площадки и не является предложением поставщика. Уточните параметры в «Экономике» и запросите КП.
          </p>
          <p className="mt-3">
            <button type="button" onClick={() => { setResult(null); setSectorId(null); setAreaRaw(''); }} className="text-sm text-muted-foreground hover:text-foreground" data-testid="start-restart">
              ← Начать заново
            </button>
          </p>
        </section>
      )}

      <AlertDialog open={overwriteOpen} onOpenChange={open => {
        setOverwriteOpen(open);
        if (!open) setPendingOverwrite(null);
      }}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Заменить сохранённый проект?</AlertDialogTitle>
            <AlertDialogDescription>
              Результат быстрого расчёта запишет новые параметры склада и заменит текущие данные этого браузера. Если хотите сохранить старый проект, сначала сохраните его ссылку или отмените замену.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel data-testid="start-overwrite-cancel">Оставить проект</AlertDialogCancel>
            <AlertDialogAction data-testid="start-overwrite-confirm" onClick={event => {
              event.preventDefault();
              if (pendingOverwrite?.destination === 'calc') finishDemoCalculation();
              else if (pendingOverwrite) finishWarehouseEstimate(pendingOverwrite.estimate);
              setPendingOverwrite(null);
              setOverwriteOpen(false);
            }}>
              Заменить и продолжить
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
